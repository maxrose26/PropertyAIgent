# Authorised finite successor scope; run1/attempt1 only.
name: Stage4 R0 completion-bridge9 ONCE
on:
  workflow_dispatch:
    inputs:
      reviewed_sha:
        required: true
        type: string
      manifest_sha:
        required: true
        type: string
      authority:
        required: true
        type: string
permissions:
  contents: read
concurrency:
  group: stage4-catchup-completion-sequential
  cancel-in-progress: false
jobs:
  source:
    runs-on: ubuntu-24.04
    timeout-minutes: 16
    steps:
      - name: Single-use binding before checkout or dependencies
        env:
          REVIEWED_SHA: ${{ inputs.reviewed_sha }}
          MANIFEST_SHA: ${{ inputs.manifest_sha }}
          SOURCE_AUTHORITY: ${{ inputs.authority }}
          GITHUB_WORKFLOW_SHA: ${{ github.workflow_sha }}
        run: |
          set -eu
          test "$GITHUB_REPOSITORY" = 'maxrose26/PropertyAIgent'
          test "$GITHUB_EVENT_NAME" = 'workflow_dispatch'
          test "$GITHUB_REF" = 'refs/heads/master'
          test "$GITHUB_SHA" = "$REVIEWED_SHA"
          test "$GITHUB_WORKFLOW_SHA" = "$REVIEWED_SHA"
          test "$GITHUB_WORKFLOW_REF" = 'maxrose26/PropertyAIgent/.github/workflows/stage4-catchup-completion-bridge9-once.yml@refs/heads/master'
          test "$GITHUB_RUN_NUMBER" = '1'
          test "$GITHUB_RUN_ATTEMPT" = '1'
          test "$SOURCE_AUTHORITY" = 'AUTHORISE_STAGE4_CATCHUP_COMPLETION_BRIDGE9_ONCE'
          test "$MANIFEST_SHA" = '0ef15c5c3bb33e91ca7146fc7cbc09e80fd56cd6d93eb8b3f354ab150ef1aac2'
          echo "SOURCE_JOB_STARTED_EPOCH=$(date +%s)" >> "$GITHUB_ENV"
      - uses: actions/checkout@v4
        with:
          ref: ${{ inputs.reviewed_sha }}
          persist-credentials: false
      - run: bash verification/ah_pg/setup.sh "$RUNNER_TEMP/r0-venv"
      - name: Existing bounded browser runtime, no council access
        run: '"$RUNNER_TEMP/r0-venv/bin/python" -m playwright install --with-deps chromium --only-shell'
      - name: Exact content and setup-budget guard before source access
        env:
          REVIEWED_SHA: ${{ inputs.reviewed_sha }}
          MANIFEST_SHA: ${{ inputs.manifest_sha }}
          SOURCE_AUTHORITY: ${{ inputs.authority }}
          GITHUB_WORKFLOW_SHA: ${{ github.workflow_sha }}
          JOB_STARTED_EPOCH: ${{ env.SOURCE_JOB_STARTED_EPOCH }}
        run: |
          "$RUNNER_TEMP/r0-venv/bin/python" - <<'PY'
          import os
          from verification.discovery.completion_source import admit
          from verification.discovery.r0_registered_scopes import checkout
          admit('bridge9',os.environ['REVIEWED_SHA'],os.environ['MANIFEST_SHA'],os.environ['SOURCE_AUTHORITY'],os.environ)
          checkout(os.environ['REVIEWED_SHA'])
          PY
      - name: One owned-process supervised fixed scope, no database
        env:
          REVIEWED_SHA: ${{ inputs.reviewed_sha }}
          MANIFEST_SHA: ${{ inputs.manifest_sha }}
          SOURCE_AUTHORITY: ${{ inputs.authority }}
          GITHUB_WORKFLOW_SHA: ${{ github.workflow_sha }}
          JOB_STARTED_EPOCH: ${{ env.SOURCE_JOB_STARTED_EPOCH }}
        run: |
          timeout --signal=TERM --kill-after=10s 505s "$RUNNER_TEMP/r0-venv/bin/python" -m verification.discovery.completion_source --scope bridge9 --reviewed-sha "$REVIEWED_SHA" --manifest-sha "$MANIFEST_SHA" --authority "$SOURCE_AUTHORITY" --output "$RUNNER_TEMP/completion-bridge9-audit"
      - name: Verify aggregate audit and identity before upload
        id: audit_gate
        if: always()
        run: |
          "$RUNNER_TEMP/r0-venv/bin/python" -m verification.discovery.completion_source --scope bridge9 --verify-audit --output "$RUNNER_TEMP/completion-bridge9-audit"
      - uses: actions/upload-artifact@v4
        if: always() && steps.audit_gate.outcome == 'success'
        with:
          name: r0-completion-bridge9-${{ github.sha }}-${{ github.run_id }}
          path: ${{ runner.temp }}/completion-bridge9-audit
          if-no-files-found: error
          retention-days: 14
