          import os
          from verification.discovery.metadata_rollout import admit
          from verification.discovery.r0_registered_scopes import checkout
          admit(os.environ['REVIEWED_SHA'],os.environ['MANIFEST_SHA'],os.environ['SOURCE_AUTHORITY'],os.environ,int(os.environ['SCOPE_INDEX']))
